GO
CREATE PROCEDURE dbo.GetProjectTasksPaginated
    @search NVARCHAR(100) = NULL,
    @orderBy NVARCHAR(100) = NULL,
    @order NVARCHAR(10) = NULL,
    @limit INT = 5,
    @offsets BIGINT = 0,
    @projectId UNIQUEIDENTIFIER = NULL,
    @statusId INT = NULL,
    @assignTo UNIQUEIDENTIFIER = NULL
AS
BEGIN
    DECLARE @searchVal NVARCHAR(100) = ISNULL(TRIM(@search) , '')
    DECLARE @tempSearch NVARCHAR(102) = CONCAT('%',@search,'%')
    select COUNT(*) 
    FROM project_tasks t 
    JOIN task_status ts on t.task_status = ts.id 
    JOIN users creation on t.created_by = creation.id
    JOIN user_profiles creation_prof on creation.id = creation_prof.[user_id] 
    LEFT JOIN users assign on assign.id = t.assign_to
    JOIN user_profiles assign_profile on assign.id = assign_profile.[user_id]
    LEFT JOIN users assign_by on assign_by.id = t.assign_by
    JOIN user_profiles assignBy_profile on assign_by.id = assignBy_profile.[user_id]
    WHERE (@search IS NULL OR t.title LIKE @tempSearch OR t.[description] LIKE @tempSearch OR ts.[status] LIKE @tempSearch) AND (@statusId IS NULL OR t.task_status = @statusId) AND (@assignTo IS NULL OR t.assign_to = @assignTo);

    SELECT
        t.*,
        ts.*,
        creation.*,
        creation_prof.*,
        assign.*,
        assign_profile.*,
        assign_by.*,
        assignBy_profile.*
    FROM project_tasks t 
    JOIN task_status ts on t.task_status = ts.id 
    JOIN users creation on t.created_by = creation.id
    JOIN user_profiles creation_prof on creation.id = creation_prof.[user_id] 
    LEFT JOIN users assign on assign.id = t.assign_to
    JOIN user_profiles assign_profile on assign.id = assign_profile.[user_id]
    LEFT JOIN users assign_by on assign_by.id = t.assign_by
    JOIN user_profiles assignBy_profile on assign_by.id = assignBy_profile.[user_id]   
    WHERE (@search IS NULL OR t.title LIKE @tempSearch OR t.[description] LIKE @tempSearch OR ts.[status] LIKE @tempSearch ) 
    AND (@statusId IS NULL OR t.task_status = @statusId)
    ORDER BY
    CASE WHEN @orderBy = 'title' AND @order = 'asc' THEN t.title END ASC,
    CASE WHEN @orderBy = 'title' AND @order = 'desc' THEN t.title END DESC,
    CASE WHEN @orderBy = 'assign' AND @order = 'asc' THEN assign.email END ASC,
    CASE WHEN @orderBy = 'assign' AND @order = 'desc' THEN assign.email END DESC,
    CASE WHEN @orderBy = 'createdBy' ANd @order = 'asc' THEN creation.email END ASC,
    CASE WHEN @orderBy = 'createdBy' ANd @order = 'desc' THEN creation.email END DESC,
    CASE WHEN @orderBy = 'assignDate' AND @order = 'asc' THEN t.assign_date END ASC,
    CASE WHEN @orderBy = 'completionDate' AND @order = 'asc' THEN t.completion_date END ASC,
    CASE WHEN @orderBy = 'completionDate' AND @order = 'desc' THEN t.completion_date END DESC
    OFFSET @offsets ROWS FETCH NEXT @limit ROWS ONLY
END
GO


drop PROCEDURE if EXISTS dbo.GetProjectTasksPaginated;

EXEC dbo.GetProjectTasksPaginated null , null , null , 5 , 0 , null , 1, null;
